cmake_minimum_required(VERSION 3.16)
project(LogSummarizer VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

# Find required packages
find_package(Qt6 COMPONENTS Core Widgets Network PrintSupport REQUIRED)
find_package(CURL REQUIRED)
find_package(JSONCPP REQUIRED)

# Include directories
include_directories(include gui)
include_directories(${CURL_INCLUDE_DIRS})
include_directories(${JSONCPP_INCLUDE_DIRS})

# GUI Sources
set(GUI_SOURCES
    src/gui_main.cpp
    gui/MainWindow.cpp
    gui/LogViewer.cpp
    gui/AnalysisPanel.cpp
    gui/SettingsDialog.cpp
    gui/AboutDialog.cpp
    src/LogParser.cpp
    src/OllamaClient.cpp
    src/SecurityAnalyzer.cpp
    src/utils.cpp
)

# CLI Sources (separate executable)
set(CLI_SOURCES
    src/main.cpp
    src/LogParser.cpp
    src/OllamaClient.cpp
    src/SecurityAnalyzer.cpp
    src/utils.cpp
)

# GUI Executable
add_executable(logsum-gui ${GUI_SOURCES})

# Link GUI libraries
target_link_libraries(logsum-gui
    Qt6::Core
    Qt6::Widgets
    Qt6::Network
    Qt6::PrintSupport
    ${CURL_LIBRARIES}
    ${JSONCPP_LIBRARIES}
)

# CLI Executable
add_executable(logsum-cli ${CLI_SOURCES})

# Link CLI libraries
target_link_libraries(logsum-cli
    ${CURL_LIBRARIES}
    ${JSONCPP_LIBRARIES}
)

# Add resources
qt_add_resources(logsum-gui "resources"
    PREFIX "/"
    FILES
        resources/icons/app_icon.png
        resources/icons/open_file.png
        resources/icons/analyze.png
        resources/icons/settings.png
        resources/icons/save.png
        resources/icons/pdf.png
        resources/icons/stop.png
        resources/icons/clear.png
        resources/icons/refresh.png
        resources/icons/test.png
        resources/icons/help.png
        resources/styles/dark_style.qss
)

# Installation
install(TARGETS logsum-gui logsum-cli DESTINATION bin)
install(DIRECTORY include/ DESTINATION include/logsummarizer)
